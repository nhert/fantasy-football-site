server {
  listen 0.0.0.0:80;
  listen 0.0.0.0:443 ssl;

  server_name  b3fl.com www.b3fl.com;

  root /usr/share/nginx/html/fantasy-football-site/dist/fantasy-football-site/browser/;

  ssl_certificate  /etc/letsencrypt/live/b3fl.com/fullchain.pem;
  ssl_certificate_key /etc/letsencrypt/live/b3fl.com/privkey.pem;

  location / {
    try_files $uri /index.csr.html;
  }
 
  location /api/ {
        # Allow local loopback addresses
        allow 127.0.0.1;
        allow ::1;

        # Deny everyone else
        deny all;

        proxy_pass http://localhost:5000; # Points to the Express PORT for the DB. This is the Node.js Express server that exists at pickems-survivor-db
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
  }

}

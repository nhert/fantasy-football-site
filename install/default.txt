server {
  listen 0.0.0.0:80;
  listen 0.0.0.0:443 ssl;

  server_name  b3fl.com www.b3fl.com;

  root /usr/share/nginx/html/fantasy-football-site/dist/fantasy-football-site/browser/;

  ssl_certificate  /etc/letsencrypt/live/b3fl.com/fullchain.pem;
  ssl_certificate_key /etc/letsencrypt/live/b3fl.com/privkey.pem;

  # Limit file uploads to 5MB max
  client_max_body_size 5M;

  location / {
    try_files $uri /index.csr.html;
  }
 
  location /api/ {
        # - the following allow/deny is for SSR
        # - Allow local loopback addresses
        # allow 127.0.0.1;
        # allow ::1;
        # - Deny everyone else
        # deny all;

        if ($http_referer !~* "^https?://(www\.)?b3fl.com") {
            return 403;
        }

        # Ensure NGINX doesn't buffer large files to disk prematurely
        proxy_request_buffering off;

        proxy_pass http://localhost:5000; # Points to the Express PORT for the DB. This is the Node.js Express server that exists at pickems-survivor-db
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
  }

  # Serve users files that have been uploaded
  location /avatars/ {
        alias /var/www/b3fl_db/pickems-survivor-db/avatar_uploads/;
        expires 1d;                # Optional: cache images in the browser for better performance
        add_header Cache-Control "public, no-transform";
        try_files $uri =404;        # Return 404 if the image file doesn't exist on disk
  }
}
